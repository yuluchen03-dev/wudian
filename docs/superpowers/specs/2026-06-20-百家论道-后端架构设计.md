### 1.2 已有辩论流程

```
init → Round1（各陈己见）→ Round2（交锋辩驳）→ 用户介入 → 会诊报告 → done
```

- Round 1：3个默认圣贤依次独立发言（各自RAG检索 → LLM生成）
- Round 2：每个圣贤看到其他人的发言后进行交锋回应
- 用户介入：指定或自动路由到一个圣贤回应
- 会诊报告：主持Agent汇总共识/分歧/建议

## 2. 架构设计方案

### 2.1 方案选择

**方案A：渐进增强（推荐）** — 在现有 `DebateEngine` 基础上扩展
- 优点：代码复用率高，风险低，迭代快
- 缺点：现有代码耦合度较高

**方案B：重构为多Agent框架** — 引入类似 LangGraph 的编排框架
- 优点：灵活性强，可扩展性好
- 缺点：引入新依赖，改造成本高

**方案C：微服务化** — 每个圣贤Agent独立服务
- 优点：完全解耦
- 缺点：对当前项目规模过度设计

**推荐方案A**：在现有代码基础上渐进增强，保持架构简洁。

### 3.2 辩论调度器 (DebateOrchestrator)

**职责**：协调主持Agent、圣贤Agent、引用校验的完整流程

```python
class DebateOrchestrator:
    """辩论调度器：协调多Agent协作"""
    
    def __init__(self):
        self.moderator = ModeratorAgent()
        self.engine = DebateEngine()  # 复用现有引擎
    
    async def orchestrate_round1(self, session_id: str) -> DebateRound:
        """协调第一轮辩论"""
        session = self.engine._get(session_id)
        rd = DebateRound(round_number=1, round_type="opinion")
        
        for sk in session.participants:
            # 1. 圣贤发言
            speech = await self._sage_speak(sk, session.topic, rd)
            
            # 2. 主持Agent检测跑题
            is_drift = await self.moderator.check_topic_drift(session.topic, speech)
            if is_drift:
                speech.content = f"（主持提醒：请围绕议题「{session.topic}」发言）\n" + speech.content
            
            rd.speeches.append(speech)
        
        return rd
```

**新增文件**: `backend/app/services/debate_orchestrator.py`

# debate_engine.py 新增方法

async def summon_guest(self, session_id: str, sage_key: str) -> SageSpeech:
    """召唤踢馆名人加入辩论"""
    session = self._get(session_id)
    
    # 1. 验证踢馆名人可用
    if sage_key not in _EXTRA_SAGES and sage_key not in SAGES:
        raise ValueError(f"不可召唤的圣贤: {sage_key}")
    
    # 2. 将踢馆名人加入参与者列表
    if sage_key not in session.participants:
        session.participants.append(sage_key)
    
    # 3. 收集所有已有发言作为上下文
    all_speeches = []
    for rd in session.rounds:
        all_speeches.extend(rd.speeches)
    
    # 4. 生成踢馆发言
    rag = get_rag_service()
    sr = await rag.retrieve_for_sage_chat(session.topic, top_k=5)
    ctx = rag.build_sage_chat_context(sr)
    msgs = _build_guest_messages(sage_key, session.topic, ctx, all_speeches)
    refs = sr.get("verses", []) + sr.get("commentaries", [])
    speech = await _speak(sage_key, msgs, session.round_counter + 1, refs)
    speech.is_guest = True
    
    # 5. 追加到当前轮次
    guest_round = DebateRound(
        round_number=session.round_counter + 1,
        round_type="guest",
        speeches=[speech],
    )
    session.rounds.append(guest_round)
    session.round_counter += 1
    
    return speech


def _build_guest_messages(sage_key, topic, rag_context, all_speeches):
    """构建踢馆名人发言的prompt"""
    sage = get_sage(sage_key)
    sys_prompt = build_sage_system_prompt(sage_key, rag_context)
    
    history = ""
    for sp in all_speeches:
        history += f"\n[{sp.sage_name}] {sp.content[:200]}\n"
    
    user_msg = f"""【百家论道 · 踢馆挑战】
议题：{topic}
已有辩论记录：{history}
你是中途加入的挑战者。请以{sage.name}身份，针对已有辩论发表你的独特见解。
要求：
1. 体现{sage.name}核心思想
2. 引用本派经典
3. 可以赞同、反对或补充任何已有观点
4. 指出回应对象
严格按JSON格式输出：（同交锋辩驳格式）"""
    
    return [{"role": "system", "content": sys_prompt}, {"role": "user", "content": user_msg}]

### 3.4 引用校验增强

```python
class CitationValidator:
    """增强版引用校验器"""
    
    def __init__(self, rag_service):
        self.rag = rag_service
    
    async def validate(self, sage_key: str, citations: List[dict], rag_refs: List[dict]) -> Tuple[List[Citation], str]:
        """校验引用，返回 (validated_citations, status)"""
        validated = []
        all_verified = True
        
        for cit in citations:
            verifiable = self._match_citation(cit, rag_refs)
            
            # 额外校验：检查引用是否属于该圣贤的经典
            if verifiable:
                belong_check = self._check_classic_belonging(sage_key, cit)
                if not belong_check:
                    verifiable = False
                    logger.warning("跨派引用: %s 引用了不属于本派的经典", sage_key)
            
            if not verifiable:
                all_verified = False
            
            validated.append(Citation(
                id=cit.get("id", ""),
                source=cit.get("出处", cit.get("source", "")),
                original=cit.get("原文", cit.get("original", "")),
                verifiable=verifiable,
            ))
        
        has_valid_quote = any(c.verifiable and c.original != "无" for c in validated)
        status = "all_verified" if all_verified else (
            "partial_verified" if has_valid_quote else "unverified"
        )
        return validated, status
    
    def _check_classic_belonging(self, sage_key: str, citation: dict) -> bool:
        """检查引用是否属于该圣贤的经典"""
        source = citation.get("出处", citation.get("source", ""))
        SAGE_CLASSICS = {
            "confucius": ["论语", "春秋", "诗经", "尚书", "礼记", "周易"],
            "mencius": ["孟子"],
            "laozi": ["道德经", "老子"],
            "mozi": ["墨子"],
            "hanfeizi": ["韩非子", "韩非"],
            "wang_yangming": ["传习录", "王阳明"],
        }
        classics = SAGE_CLASSICS.get(sage_key, [])
        if not source or source == "无":
            return True
        return any(c in source for c in classics)
```

### 3.5 辩论持久化

```python
class DebateStorage:
    """辩论会话持久化（文件存储）"""
    
    def __init__(self, storage_dir: str = "data/debates"):
        self.storage_dir = Path(storage_dir)
        self.storage_dir.mkdir(parents=True, exist_ok=True)
        self._cache: Dict[str, DebateSession] = {}
    
    def save(self, session: DebateSession) -> None:
        self._cache[session.id] = session
        path = self.storage_dir / f"{session.id}.json"
        with open(path, "w", encoding="utf-8") as f:
            f.write(session.model_dump_json(indent=2))
    
    def load(self, session_id: str) -> Optional[DebateSession]:
        if session_id in self._cache:
            return self._cache[session_id]
        path = self.storage_dir / f"{session_id}.json"
        if path.exists():
            with open(path, "r", encoding="utf-8") as f:
                data = json.load(f)
            session = DebateSession(**data)
            self._cache[session_id] = session
            return session
        return None
```

## 7. 风险与应对

| 风险 | 影响 | 应对 |
|------|------|------|
| 多Agent LLM调用延迟高 | 辩论等待时间长 | 流式输出 + 轮流呈现（将延迟转化为节奏） |
| LLM输出格式不稳定 | JSON解析失败 | 已有 `_parse_json` 容错 + fallback |
| 引用编造 | 破坏信任感 | 强制RAG引用校验 + 诚实降级 |
| 圣贤人格串台 | 用户体验差 | 每个Agent独立system prompt + 经典归属校验 |
| 内存会话丢失 | 用户进度丢失 | 持久化到文件系统 |
