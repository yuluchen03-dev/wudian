```python
# backend/app/services/debate_engine.py:166-225
async def summon_guest(self, session_id: str, sage_key: str) -> SageSpeech:
    """召唤踢馆名人加入辩论"""
    session = self._get(session_id)
    
    # 1. 验证踢馆名人可用
    if sage_key not in _EXTRA_SAGES and sage_key not in SAGES:
        raise ValueError(f"不可召唤的圣贤: {sage_key}")
    
    # 2. 将踢馆名人加入参与者列表
    if sage_key not in session.participants:
        session.participants.append(sage_key)
    
    # 3. 收集所有已有发言作为上下文
    all_speeches = []
    for rd in session.rounds:
        all_speeches.extend(rd.speeches)
    
    # 4. 生成踢馆发言
    rag = get_rag_service()
    sr = await rag.retrieve_for_sage_chat(session.topic, top_k=5)
    ctx = rag.build_sage_chat_context(sr)
    msgs = _build_guest_messages(sage_key, session.topic, ctx, all_speeches)
    refs = sr.get("verses", []) + sr.get("commentaries", [])
    speech = await _speak(sage_key, msgs, session.round_counter + 1, refs)
    speech.is_guest = True
    
    # 5. 追加到当前轮次
    guest_round = DebateRound(
        round_number=session.round_counter + 1,
        round_type="guest",
        speeches=[speech],
    )
    session.rounds.append(guest_round)
    session.round_counter += 1
    
    return speech


def _build_guest_messages(sage_key, topic, rag_context, all_speeches):
    """构建踢馆名人发言的prompt"""
    sage = get_sage(sage_key)
    sys_prompt = build_sage_system_prompt(sage_key, rag_context)
    
    history = ""
    for sp in all_speeches:
        history += f"\n[{sp.sage_name}] {sp.content[:200]}\n"
    
    user_msg = f"""【百家论道 · 踢馆挑战】
议题：{topic}
已有辩论记录：{history}
你是中途加入的挑战者。请以{sage.name}身份，针对已有辩论发表你的独特见解。
要求：
1. 体现{sage.name}核心思想
2. 引用本派经典
3. 可以赞同、反对或补充任何已有观点
4. 指出回应对象
严格按JSON格式输出：（同交锋辩驳格式）"""
    
    return [{"role": "system", "content": sys_prompt}, {"role": "user", "content": user_msg}]
```

```python
# backend/app/services/moderator_agent.py
"""
悟典 - 百家论道主持Agent

职责：
- 路由用户介入到合适的圣贤
- 检测发言是否跑题
- 判断辩论是否应继续
- 生成会诊报告
"""
import logging
from typing import List, Optional

from app.schemas.debate import DebateSession, SageSpeech, DebateRound
from app.services.llm_client import llm_client
from app.services.sage_persona import SAGES

logger = logging.getLogger(__name__)

# 跑题检测提示词
TOPIC_DRIFT_SYSTEM = """你是一个辩论主持人，负责判断发言是否偏离议题。

判断标准：
1. 发言是否与议题直接相关
2. 是否在讨论议题的核心问题
3. 是否有实质性内容而非空泛议论

请以JSON格式返回：{"is_drift": true/false, "reason": "判断理由"}"""

# 继续判断提示词
SHOULD_CONTINUE_SYSTEM = """你是一个辩论主持人，判断辩论是否应继续。

判断标准：
1. 是否还有新的观点可以提出
2. 是否陷入了重复论证
3. 是否达到了有价值的结论

请以JSON格式返回：{"should_continue": true/false, "reason": "判断理由"}"""


class ModeratorAgent:
    """主持Agent：控场与路由"""
    
    async def check_topic_drift(self, topic: str, speech: SageSpeech) -> bool:
        """跑题检测：判断圣贤发言是否偏离议题"""
        messages = [
            {"role": "system", "content": TOPIC_DRIFT_SYSTEM},
            {"role": "user", "content": f"议题：{topic}\n发言：{speech.content}\n是否跑题？"},
        ]
        result = await llm_client.complete_json(messages, temperature=0.1)
        return result.get("is_drift", False)
    
    async def route_intervention(self, session: DebateSession, content: str) -> str:
        """路由用户介入到最合适的圣贤"""
        # 优先：用户指定的目标
        # 其次：基于发言内容的关键词匹配
        # 最后：最后一个发言的圣贤
        
        # 简单实现：返回最后一个发言的圣贤
        if session.rounds:
            last_round = session.rounds[-1]
            if last_round.speeches:
                return last_round.speeches[-1].sage_key
        
        # 如果没有发言记录，返回第一个参与者
        return session.participants[0] if session.participants else "confucius"
    
    async def should_continue(self, session: DebateSession) -> bool:
        """判断辩论是否应继续（防死循环）"""
        # 简单实现：最多3轮
        if session.round_counter >= 3:
            return False
        
        # 收集最近两轮的发言
        if len(session.rounds) < 2:
            return True
        
        recent_rounds = session.rounds[-2:]
        all_speeches = []
        for rd in recent_rounds:
            all_speeches.extend(rd.speeches)
        
        if not all_speeches:
            return True
        
        # 检查发言内容是否有实质性变化
        messages = [
            {"role": "system", "content": SHOULD_CONTINUE_SYSTEM},
            {"role": "user", "content": f"最近发言：\n{chr(10).join([s.content[:200] for s in all_speeches])}"},
        ]
        result = await llm_client.complete_json(messages, temperature=0.1)
        return result.get("should_continue", True)


# 全局实例
moderator_agent = ModeratorAgent()
```

```python
# backend/app/services/debate_orchestrator.py
"""
悟典 - 百家论道辩论调度器

职责：
- 协调主持Agent、圣贤Agent、引用校验的完整流程
- 管理辩论轮次流转
- 集成跑题检测和死循环防护
"""
import logging
from typing import List, Optional

from app.schemas.debate import DebateSession, DebateRound, SageSpeech
from app.services.debate_engine import debate_engine, _speak, _build_opinion_messages, _build_cross_messages
from app.services.moderator_agent import moderator_agent
from app.services.rag_service import get_rag_service
from app.services.sage_persona import get_sage, build_sage_system_prompt

logger = logging.getLogger(__name__)


class DebateOrchestrator:
    """辩论调度器：协调多Agent协作"""
    
    def __init__(self):
        self.engine = debate_engine
        self.moderator = moderator_agent
    
    async def orchestrate_round1(self, session_id: str) -> DebateRound:
        """协调第一轮辩论（各陈己见）"""
        session = self.engine._get(session_id)
        rd = DebateRound(round_number=1, round_type="opinion")
        
        rag = get_rag_service()
        
        for sk in session.participants:
            # 1. 圣贤发言
            sr = await rag.retrieve_for_sage_chat(session.topic, top_k=5)
            ctx = rag.build_sage_chat_context(sr)
            msgs = _build_opinion_messages(sk, session.topic, ctx)
            refs = sr.get("verses", []) + sr.get("commentaries", [])
            speech = await _speak(sk, msgs, 1, refs)
            
            # 2. 主持Agent检测跑题
            is_drift = await self.moderator.check_topic_drift(session.topic, speech)
            if is_drift:
                speech.content = f"（主持提醒：请围绕议题「{session.topic}」发言）\n" + speech.content
            
            rd.speeches.append(speech)
        
        return rd
    
    async def orchestrate_round2(self, session_id: str) -> DebateRound:
        """协调第二轮辩论（交锋辩驳）"""
        session = self.engine._get(session_id)
        rd = DebateRound(round_number=2, round_type="cross_debate")
        
        # 获取第一轮的发言作为上下文
        round1 = session.rounds[0] if session.rounds else None
        if not round1:
            raise ValueError("没有第一轮辩论记录")
        
        rag = get_rag_service()
        
        for sk in session.participants:
            # 1. 圣贤发言
            sr = await rag.retrieve_for_sage_chat(session.topic, top_k=5)
            ctx = rag.build_sage_chat_context(sr)
            msgs = _build_cross_messages(sk, session.topic, ctx, round1.speeches)
            refs = sr.get("verses", []) + sr.get("commentaries", [])
            speech = await _speak(sk, msgs, 2, refs)
            
            # 2. 主持Agent检测跑题
            is_drift = await self.moderator.check_topic_drift(session.topic, speech)
            if is_drift:
                speech.content = f"（主持提醒：请围绕议题「{session.topic}」发言）\n" + speech.content
            
            rd.speeches.append(speech)
        
        return rd
    
    async def orchestrate_intervention(self, session_id: str, content: str, target_sage: Optional[str] = None) -> SageSpeech:
        """协调用户介入"""
        session = self.engine._get(session_id)
        
        # 如果没有指定目标圣贤，由主持Agent路由
        if not target_sage:
            target_sage = await self.moderator.route_intervention(session, content)
        
        # 生成介入回应
        rag = get_rag_service()
        sr = await rag.retrieve_for_sage_chat(session.topic, top_k=5)
        ctx = rag.build_sage_chat_context(sr)
        
        # 收集所有发言作为上下文
        all_speeches = []
        for rd in session.rounds:
            all_speeches.extend(rd.speeches)
        
        msgs = _build_intervention_messages(target_sage, session.topic, ctx, all_speeches, content)
        refs = sr.get("verses", []) + sr.get("commentaries", [])
        speech = await _speak(target_sage, msgs, session.round_counter + 1, refs)
        
        return speech
    
    async def check_should_continue(self, session_id: str) -> bool:
        """检查辩论是否应继续"""
        session = self.engine._get(session_id)
        return await self.moderator.should_continue(session)
```

```python
# backend/app/services/citation_validator.py
"""
悟典 - 百家论道增强版引用校验器

职责：
- 校验引用的真实性
- 检查引用是否属于该圣贤的经典
- 提供引用状态反馈
"""
import logging
from typing import List, Tuple, Dict, Any

from app.schemas.debate import Citation
from app.services.rag_service import get_rag_service

logger = logging.getLogger(__name__)

# 圣贤经典映射
SAGE_CLASSICS = {
    "confucius": ["论语", "春秋", "诗经", "尚书", "礼记", "周易"],
    "mencius": ["孟子"],
    "laozi": ["道德经", "老子"],
    "mozi": ["墨子"],
    "hanfeizi": ["韩非子", "韩非"],
    "wang_yangming": ["传习录", "王阳明"],
}


class CitationValidator:
    """增强版引用校验器"""
    
    def __init__(self):
        self.rag = get_rag_service()
    
    async def validate(self, sage_key: str, citations: List[dict], rag_refs: List[dict]) -> Tuple[List[Citation], str]:
        """校验引用，返回 (validated_citations, status)"""
        validated = []
        all_verified = True
        
        for cit in citations:
            verifiable = self._match_citation(cit, rag_refs)
            
            # 额外校验：检查引用是否属于该圣贤的经典
            if verifiable:
                belong_check = self._check_classic_belonging(sage_key, cit)
                if not belong_check:
                    verifiable = False
                    logger.warning("跨派引用: %s 引用了不属于本派的经典", sage_key)
            
            if not verifiable:
                all_verified = False
            
            validated.append(Citation(
                id=cit.get("id", ""),
                source=cit.get("出处", cit.get("source", "")),
                original=cit.get("原文", cit.get("original", "")),
                verifiable=verifiable,
            ))
        
        has_valid_quote = any(c.verifiable and c.original != "无" for c in validated)
        status = "all_verified" if all_verified else (
            "partial_verified" if has_valid_quote else "unverified"
        )
        return validated, status
    
    def _match_citation(self, citation: dict, rag_refs: List[dict]) -> bool:
        """匹配引用与RAG参考"""
        cit_original = citation.get("原文", citation.get("original", ""))
        if not cit_original or cit_original == "无":
            return False
        
        # 检查是否与RAG参考匹配
        for ref in rag_refs:
            ref_original = ref.get("original", "")
            if not ref_original:
                continue
            
            # 子串匹配
            if cit_original in ref_original or ref_original in cit_original:
                return True
            
            # 清理标点后匹配
            cleaned_cit = cit_original.replace("。", "").replace("，", "").replace("？", "").replace("！", "").replace("；", "").strip()
            cleaned_ref = ref_original.replace("。", "").replace("，", "").replace("？", "").replace("！", "").replace("；", "").strip()
            
            if cleaned_cit in cleaned_ref or cleaned_ref in cleaned_cit:
                return True
        
        return False
    
    def _check_classic_belonging(self, sage_key: str, citation: dict) -> bool:
        """检查引用是否属于该圣贤的经典"""
        source = citation.get("出处", citation.get("source", ""))
        classics = SAGE_CLASSICS.get(sage_key, [])
        if not source or source == "无":
            return True
        return any(c in source for c in classics)
```

```python
# backend/app/services/debate_storage.py
"""
悟典 - 百家论道辩论持久化存储

职责：
- 持久化辩论会话到文件系统
- 提供会话加载和缓存
- 支持会话历史查询
"""
import json
import logging
from pathlib import Path
from typing import Dict, Optional, List

from app.schemas.debate import DebateSession

logger = logging.getLogger(__name__)


class DebateStorage:
    """辩论会话持久化（文件存储）"""
    
    def __init__(self, storage_dir: str = "data/debates"):
        self.storage_dir = Path(storage_dir)
        self.storage_dir.mkdir(parents=True, exist_ok=True)
        self._cache: Dict[str, DebateSession] = {}
    
    def save(self, session: DebateSession) -> None:
        """保存会话到文件"""
        self._cache[session.id] = session
        path = self.storage_dir / f"{session.id}.json"
        with open(path, "w", encoding="utf-8") as f:
            f.write(session.model_dump_json(indent=2))
        logger.info(f"保存会话: {session.id}")
    
    def load(self, session_id: str) -> Optional[DebateSession]:
        """加载会话"""
        if session_id in self._cache:
            return self._cache[session_id]
        
        path = self.storage_dir / f"{session_id}.json"
        if path.exists():
            with open(path, "r", encoding="utf-8") as f:
                data = json.load(f)
            session = DebateSession(**data)
            self._cache[session_id] = session
            logger.info(f"加载会话: {session_id}")
            return session
        
        return None
    
    def list_sessions(self, limit: int = 20) -> List[DebateSession]:
        """列出最近的会话"""
        sessions = []
        for path in sorted(self.storage_dir.glob("*.json"), reverse=True)[:limit]:
            try:
                with open(path, "r", encoding="utf-8") as f:
                    data = json.load(f)
                sessions.append(DebateSession(**data))
            except Exception as e:
                logger.error(f"加载会话失败 {path}: {e}")
        return sessions
    
    def delete(self, session_id: str) -> bool:
        """删除会话"""
        if session_id in self._cache:
            del self._cache[session_id]
        
        path = self.storage_dir / f"{session_id}.json"
        if path.exists():
            path.unlink()
            logger.info(f"删除会话: {session_id}")
            return True
        
        return False


# 全局实例
debate_storage = DebateStorage()
```
